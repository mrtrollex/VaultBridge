// ADR 0011: this journal gates cooperating UI only; it never authorizes REST.
export const LOCK = "vaultbridge.named-ui-session.v1";
export const INTENT = "vaultbridge.named-ui-session.intent.v1";
export const JOURNAL = "vaultbridge.named-ui-session.journal.v1";
const dirty = new Set(["posting", "deleting", "uncertain"]);
const phases = new Set(["locked", "restorable", ...dirty, "logout_pending"]);
const nonce = () => Array.from(crypto.getRandomValues(new Uint8Array(16)), n => n.toString(16).padStart(2, "0")).join("");
const isNonce = value => typeof value === "string" && /^[0-9a-f]{32}$/.test(value);
const equal = (a, b) => JSON.stringify(a) === JSON.stringify(b);

export class NamedSession {
  constructor(url, changed) {
    this.url = url;
    this.changed = changed;
    this.epoch = 0;
    this.snapshot = null;
    this.reads = new Set();
    this.supported = window.isSecureContext && window.top === window.self && !!navigator.locks?.request;
    try { this.records(); } catch { this.supported = false; }
    this.state = this.supported ? "locked" : "unsupported";
    window.addEventListener("storage", e => {
      if (e.key === null || e.key === INTENT || e.key === JOURNAL) this.check();
    });
    window.addEventListener("pagehide", () => this.blank("locked"));
    document.addEventListener("visibilitychange", () => {
      if (document.hidden) this.blank("locked");
      else this.resume();
    });
    window.addEventListener("pageshow", () => this.resume());
    window.addEventListener("focus", () => this.resume());
  }

  resume() {
    if (this.mutating) return;
    try {
      const journal = this.records().journal;
      // A first-visit locked form has no session to restore. In particular,
      // focusing that form must not erase a key the user is entering.
      if (this.snapshot || (journal && journal.phase !== "locked")) void this.restore();
    } catch { this.blank("unsupported"); }
  }

  read(key) {
    const raw = localStorage.getItem(key);
    if (raw === null) return null;
    if (new TextEncoder().encode(raw).length > 512) throw Error("journal");
    const value = JSON.parse(raw);
    if (!value || typeof value !== "object" || Array.isArray(value)) throw Error("journal");
    return value;
  }

  records() {
    const intent = this.read(INTENT);
    const journal = this.read(JOURNAL);
    if (intent && (Object.keys(intent).sort().join() !== "nonce,version" || intent.version !== 1 || !isNonce(intent.nonce))) throw Error("journal");
    if (journal && (journal.version !== 1 || !phases.has(journal.phase) || !isNonce(journal.revision) ||
        !(journal.intent === null || isNonce(journal.intent)) ||
        (journal.mutation !== undefined && !isNonce(journal.mutation)) ||
        Object.keys(journal).some(k => !["version", "intent", "phase", "revision", "mutation"].includes(k)))) throw Error("journal");
    return {intent: intent?.nonce ?? null, journal};
  }

  write(phase, intent, mutation) {
    const journal = {version: 1, intent, phase, revision: nonce(), ...(mutation ? {mutation} : {})};
    localStorage.setItem(JOURNAL, JSON.stringify(journal));
    if (!equal(this.records().journal, journal)) throw Error("journal");
    return journal;
  }

  blank(state) {
    this.epoch++;
    this.snapshot = null;
    for (const controller of this.reads) controller.abort();
    this.reads.clear();
    this.state = state;
    this.changed(state);
  }

  check(snapshot = this.snapshot) {
    if (!this.supported) return false;
    try {
      const record = this.records();
      if (snapshot && record.intent === snapshot.intent && equal(record.journal, snapshot) && snapshot.phase === "restorable") return true;
      if (this.snapshot) this.blank(dirty.has(record.journal?.phase) ? "pending" : "logout_pending");
    } catch { this.blank("unsupported"); }
    return false;
  }

  async locked(action) {
    if (!this.supported) { this.blank("unsupported"); return; }
    try { return await navigator.locks.request(LOCK, {mode: "exclusive"}, action); }
    catch { this.blank(this.mutating ? "uncertain" : "unsupported"); }
  }

  async wire(method, body, signal) {
    const response = await fetch(this.url, {
      method, credentials: "same-origin", cache: "no-store", redirect: "error", signal,
      headers: {"X-VaultBridge-UI-Request": "1", ...(body ? {"Content-Type": "application/json"} : {})},
      ...(body ? {body} : {}),
    });
    // A complete body is part of settlement. Never abort cookie mutations.
    const data = await response.json();
    if (!data || typeof data !== "object" || Array.isArray(data)) throw Error("response");
    if (response.ok && (Object.keys(data).join() !== "authenticated" || data.authenticated !== (method !== "DELETE"))) throw Error("response");
    const retry = response.headers.get("Retry-After");
    this.retryAfter = retry && /^[0-9]{1,5}$/.test(retry) ? Number(retry) : null;
    return response;
  }

  async login(key) {
    if (!this.supported) { this.blank("unsupported"); return; }
    let captured;
    try {
      const {intent, journal} = this.records();
      captured = intent;
      // A submission during pending deletion is not a deliberate login AFTER
      // confirmed logout. Never let waiting for the lock turn it into one.
      if (journal?.phase === "uncertain") { this.blank("uncertain"); return; }
      if (journal?.phase === "deleting" || journal?.phase === "logout_pending" ||
          (journal && journal.intent !== intent) || (!journal && intent !== null)) {
        this.blank("logout_pending");
        return;
      }
    } catch { this.blank("unsupported"); return; }
    this.blank("pending");
    await this.locked(async () => {
      let record = this.records();
      if (record.intent !== captured) { this.blank("locked"); return; }
      if (dirty.has(record.journal?.phase)) { this.blank("uncertain"); return; }
      if (record.journal && record.journal.intent !== captured) { this.blank("logout_pending"); return; }
      if (record.journal?.phase === "logout_pending") { this.blank("logout_pending"); return; }
      this.mutating = true;
      let sent = false;
      try {
        this.write("posting", captured, nonce());
        sent = true;
        const response = await this.wire("POST", JSON.stringify({api_key: key}));
        key = "";
        // 5xx is conservatively ambiguous, even if its body was received.
        if (response.status >= 500) throw Error("outcome");
        const journal = this.write(response.ok ? "restorable" : "locked", captured);
        if (this.records().intent !== captured) { this.blank("locked"); return; }
        if (document.hidden) { this.blank("locked"); return; }
        if (response.ok) { this.snapshot = journal; this.state = "authenticated"; this.changed(this.state); }
        else this.blank(response.status === 429 ? "rate_limited" : "rejected");
      } catch {
        if (!sent) {
          // No issuer was dispatched. Unwritable required browser storage is
          // unsupported, not evidence of an abandoned HTTP response.
          this.supported = false;
          this.blank("unsupported");
          return;
        }
        // The durable posting record remains dirty if writing uncertainty fails.
        try { this.write("uncertain", captured); } catch { /* fail closed */ }
        this.blank("uncertain");
      } finally { key = ""; this.mutating = false; }
    });
    key = "";
  }

  async restore() {
    if (!this.supported) { this.blank("unsupported"); return; }
    this.blank("checking");
    const epoch = this.epoch;
    const snapshot = await this.locked(() => {
      const {intent, journal} = this.records();
      if (dirty.has(journal?.phase)) { this.blank("uncertain"); return null; }
      if (!journal || journal.intent !== intent || journal.phase !== "restorable") {
        this.blank(journal?.phase === "logout_pending" ? "logout_pending" : "locked");
        return null;
      }
      // Detect write denial without modifying the revision or granting access.
      localStorage.setItem(JOURNAL, JSON.stringify(journal));
      return journal;
    });
    if (!snapshot || epoch !== this.epoch) return;
    const controller = new AbortController();
    this.reads.add(controller);
    try {
      const response = await this.wire("GET", null, controller.signal);
      if (epoch !== this.epoch) return;
      if (!this.check(snapshot)) {
        if (epoch === this.epoch) {
          const {intent, journal} = this.records();
          this.blank(dirty.has(journal?.phase) ? "uncertain" :
            journal?.phase === "logout_pending" || (journal && journal.intent !== intent) ? "logout_pending" : "locked");
        }
        return;
      }
      if (response.ok) { this.snapshot = snapshot; this.state = "authenticated"; this.changed(this.state); }
      else if (response.status === 401) await this.unauthorized(snapshot);
      else this.blank(response.status === 429 ? "rate_limited" : "unavailable");
    } catch { if (epoch === this.epoch) this.blank("unavailable"); }
    finally { this.reads.delete(controller); }
  }

  async logout() {
    this.blank("pending");
    if (!this.supported) { this.blank("unsupported"); return; }
    try { localStorage.setItem(INTENT, JSON.stringify({version: 1, nonce: nonce()})); }
    catch { this.blank("uncertain"); return; }
    await this.locked(() => this.remove());
  }

  async unauthorized(snapshot) {
    // Compare under the mutation lock BEFORE publishing intent or clearing a
    // newer session. GET/read responses themselves never delete cookies.
    if (!this.check(snapshot)) return;
    this.blank("pending");
    await this.locked(async () => {
      if (!this.check(snapshot)) return;
      this.blank("pending");
      localStorage.setItem(INTENT, JSON.stringify({version: 1, nonce: nonce()}));
      await this.remove();
    });
  }

  async remove() {
    const {intent, journal} = this.records();
    if (dirty.has(journal?.phase)) { this.blank("uncertain"); return; }
    this.mutating = true;
    try {
      this.write("deleting", intent, nonce());
      const response = await this.wire("DELETE");
      if (response.status >= 500) throw Error("outcome");
      this.write(response.ok ? "locked" : "logout_pending", intent);
      if (this.records().intent !== intent) { this.blank("logout_pending"); return; }
      this.blank(response.ok ? "logged_out" : "logout_pending");
    } catch {
      try { this.write("uncertain", intent); } catch { /* retain dirty evidence */ }
      this.blank("uncertain");
    } finally { this.mutating = false; }
  }
}

export class ScopedClient {
  constructor(session, base) { this.session = session; this.base = base; }

  async get(path, params, signal) {
    const snapshot = this.session.snapshot;
    const epoch = this.session.epoch;
    if (!this.session.check(snapshot)) throw Error("session");
    const url = new URL(path, this.base);
    if (url.origin !== location.origin) throw Error("origin");
    url.search = new URLSearchParams(params).toString();
    const response = await fetch(url, {signal, cache: "no-store", credentials: "same-origin", redirect: "error",
      headers: {"X-VaultBridge-UI-Request": "1"}});
    // Authentication failure is authoritative even when its body is unreadable.
    // Keep the captured session guards before the coordinated, synchronous blank.
    if (response.status === 401) {
      if (epoch !== this.session.epoch || !this.session.check(snapshot)) throw Error("stale");
      await this.session.unauthorized(snapshot); throw Error("session");
    }
    const data = await response.json();
    if (epoch !== this.session.epoch || !this.session.check(snapshot)) throw Error("stale");
    if (!response.ok) {
      const error = Error(({403: "denied", 429: "rate_limited", 503: "unavailable"})[response.status] || "unavailable");
      const retry = response.headers.get("Retry-After");
      if (retry && /^[0-9]{1,5}$/.test(retry)) error.retryAfter = Number(retry);
      throw error;
    }
    return data;
  }
}

import re

from fastapi import APIRouter, HTTPException, Request
from fastapi.responses import FileResponse, JSONResponse

from app.api.dependencies import enforce_rate_limit

from app.core.ui_session import UI_SESSION_COOKIE_NAME, guard_named_ui_request, validate_ui_session_token
from app.ui.router import (
    _ASSET_ROOT,
    _UI_ROOT,
    UI_SECURITY_HEADERS,
    UISessionRequest,
    _clear_session_cookie,
    _set_session_cookie,
    redirect_to_dashboard,
)

NAMED_ASSETS = {
    "app.css": "text/css", "vaultbridge-logo.webp": "image/webp",
    "named-app.js": "text/javascript", "named-session.js": "text/javascript",
    "scoped-client.js": "text/javascript", "spaces.js": "text/javascript",
}
NAMED_UI_PATHS = frozenset({"/ui", "/ui/", "/ui/session"} | {
    "/ui/assets/" + name for name in NAMED_ASSETS
})
router = APIRouter(include_in_schema=False)
HEADERS = {**UI_SECURITY_HEADERS, "Cache-Control": "no-store"}


@router.api_route("/ui/session", methods=["POST", "GET", "DELETE"])
async def session(request: Request):
    settings = request.app.state.settings
    try:
        enforce_rate_limit(request, settings, request.app.state.rate_limiter)
        guard_named_ui_request(request)
        if request.method == "POST":
            types = request.headers.getlist("content-type")
            if len(types) != 1 or not re.fullmatch(
                    r'application/json(?:\s*;\s*charset=(?:[a-z0-9._-]+|"[a-z0-9._-]+"))?\s*',
                    types[0], re.IGNORECASE):
                raise HTTPException(415, "invalid_session_request")
            try:
                payload = UISessionRequest.model_validate_json(await request.body())
            except ValueError:
                raise HTTPException(422, "Invalid session request") from None
            verify_api_key(payload.api_key.get_secret_value(), settings)
            response = JSONResponse({"authenticated": True}, headers=HEADERS)
            _set_session_cookie(response, request, settings)
            return response
        if request.method == "DELETE":
            response = JSONResponse({"authenticated": False}, headers=HEADERS)
            _clear_session_cookie(response, request)
            return response
        if not validate_ui_session_token(request.cookies.get(UI_SESSION_COOKIE_NAME), settings):
            raise HTTPException(401, "Authentication required")
        return JSONResponse({"authenticated": True}, headers=HEADERS)
    except HTTPException as exc:
        return JSONResponse({"detail": exc.detail}, status_code=exc.status_code,
                            headers={**(exc.headers or {}), **HEADERS})


router.add_api_route("/ui", redirect_to_dashboard, methods=["GET", "HEAD"])

def dashboard():
    return FileResponse(_UI_ROOT / "named.html", media_type="text/html", headers=HEADERS)

    def serve():
        return FileResponse(_ASSET_ROOT / name, media_type=media_type, headers=UI_SECURITY_HEADERS)
    return serve

class Barrier:
    def __init__(self):
        self.entered = threading.Event()
        self.release = threading.Event()
        self.method = None
        self.headers = None
        self.failure = None
        self.path = "/ui/session"
        self.stage = "headers"
        self.truncate = False
        self.finished = threading.Event()
        self.repeat = False
        self.captured_headers = []

    def arm(self, method, failure=None, path="/ui/session", *, stage="headers", truncate=False, repeat=False):
        self.entered.clear()
        self.release.clear()
        self.method = method
        self.failure = failure
        self.headers = None
        self.path = path
        self.stage = stage
        self.truncate = truncate
        self.finished.clear()
        self.repeat = repeat
        self.captured_headers.clear()

    async def __call__(self, app, scope, receive, send):
        held = (
            self.method is not None
            and scope["path"].endswith(self.path)
            and scope["method"] == self.method
            and (not self.repeat or not self.release.is_set())
        )
        failure = self.failure if held else None
        if held and not self.repeat:
            self.method = None

        async def intercept(message):
            if held and message["type"] == "http.response.start":
                self.headers = dict(message["headers"])
                self.captured_headers.append(self.headers)
            stopping = held and (
                (self.stage == "headers" and message["type"] == "http.response.start")
                or (self.stage == "body" and message["type"] == "http.response.body")
            )
            if stopping:
                self.entered.set()
                assert await asyncio.to_thread(self.release.wait, 20), "Test did not release response headers"
                if self.truncate and message["type"] == "http.response.body":
                    await send({**message, "body": message["body"][:1], "more_body": True})
                    raise RuntimeError("Synthetic truncated HTTP response")
            await send(message)

        try:
            if failure is not None:
                from starlette.responses import JSONResponse

                await JSONResponse(
                    {"detail": "test_failure"}, status_code=failure, headers={"Cache-Control": "no-store"}
                )(scope, receive, intercept)
            else:
                await app(scope, receive, intercept)
        finally:
            if held:
                self.finished.set()


class _Tunnel(socketserver.StreamRequestHandler):
    # Never buffer TLS bytes sent immediately after CONNECT headers: bytes
    # prefetched by rfile would be invisible to the socket forwarding loop.
    rbufsize = 0

    def handle(self):
        with self.server.mutex:
            if self.server.failed:
                return
        first = self.rfile.readline().decode("ascii").strip().split()
        if len(first) != 3:
            return
        connect = first[0] == "CONNECT"
        target = urlsplit(first[1]) if not connect else None
        host, port = first[1].rsplit(":", 1) if connect else (target.hostname, target.port or 80)
        if host not in {"vaultbridge.example.test", "other.example.test", "foreign.invalid.test"}:
            return
        headers = []
        while (line := self.rfile.readline()) not in {b"\r\n", b"\n", b""}:
            if line.split(b":", 1)[0].lower() not in {b"connection", b"proxy-connection"}:
                headers.append(line)
        with socket.create_connection(("127.0.0.1", int(port)), timeout=5) as peer:
            if connect:
                self.wfile.write(b"HTTP/1.1 200 Connection established\r\n\r\n")
                self.wfile.flush()
            else:
                assert target.scheme == "http"
                path = target.path or "/"
                if target.query:
                    path += "?" + target.query
                peer.sendall(
                    f"{first[0]} {path} {first[2]}\r\n".encode() + b"".join(headers) + b"Connection: close\r\n\r\n"
                )
            self.connection.settimeout(20)
            peer.settimeout(20)
            with self.server.mutex:
                self.server.tunnels.add(self.connection)
            try:
                while True:
                    ready, _, _ = select.select([self.connection, peer], [], [], 20)
                    if not ready:
                        return
                    for source in ready:
                        try:
                            data = source.recv(65536)
                            if not data:
                                return
                            (peer if source is self.connection else self.connection).sendall(data)
                        except OSError:
                            return
            finally:
                with self.server.mutex:
                    self.server.tunnels.discard(self.connection)


class _Proxy(socketserver.ThreadingTCPServer):
    allow_reuse_address = True
    daemon_threads = True

    def __init__(self, *args):
        self.mutex = threading.Lock()
        self.tunnels = set()
        self.failed = False
        super().__init__(*args)

    def disconnect(self):
        """Actual TCP failure; no browser fetch/response mock."""
        with self.mutex:
            self.failed = True
            for connection in self.tunnels:
                try:
                    connection.shutdown(socket.SHUT_RDWR)
                except OSError:
                    pass

    def recover(self):
        with self.mutex:
            self.failed = False

@contextmanager
def serve(app, cert=None, key=None):
    listener = socket.socket()
    listener.bind(("127.0.0.1", 0))
    listener.listen()
    port = listener.getsockname()[1]
    ready = threading.Event()

    class ReadyServer(uvicorn.Server):
        async def startup(self, sockets=None):
            await super().startup(sockets=sockets)
            if self.started:
                ready.set()

    server = ReadyServer(
        uvicorn.Config(
            app,
            access_log=False,
            log_level="error",
            lifespan="on",
            ssl_certfile=str(cert) if cert else None,
            ssl_keyfile=str(key) if key else None,
            proxy_headers=False,
        )
    )
    thread = threading.Thread(target=server.run, kwargs={"sockets": [listener]}, daemon=True)
    thread.start()
    try:
        assert ready.wait(10), "Disposable server startup did not complete"
        yield port
    finally:
        server.should_exit = True
        thread.join(10)
        listener.close()
        assert not thread.is_alive(), "Disposable TLS server did not stop"

def observe_protected_reads(page):
    # Observe completion and actual controller memory using the real modules;
    # do not replace their session, parsing, authorization or rendering behavior.
    page.evaluate("""async () => {
      const {ScopedClient}=await import(new URL('./assets/scoped-client.js', document.baseURI));
      const {Spaces}=await import(new URL('./assets/spaces.js', document.baseURI));
      window.protectedReads=[];
      window.protectedResponses=[];
      const nativeFetch=window.fetch;
      window.fetch=(...args) => nativeFetch(...args).then(response => {
        if (new URL(response.url).pathname.includes('/api/v1/knowledge/spaces'))
          protectedResponses.push(response.status);
        return response;
      });
      const get=ScopedClient.prototype.get, reset=Spaces.prototype.reset;
      ScopedClient.prototype.get=async function(...args) {
        const row={done:false}; protectedReads.push(row);
        try { return await get.apply(this,args); } finally { row.done=true; }
      };
      Spaces.prototype.reset=function() {
        reset.call(this);
        window.erasedPrivate={choices:this.choices.length, selected:this.selected.length};
      };
      AbortController.prototype.abort=function() {};
    }""")

def fulfill_protected_401(route, body_kind):
    body, content_type = PROTECTED_401_BODIES[body_kind]
    route.fulfill(status=401, body=body, content_type=content_type, headers={"Cache-Control": "no-store"})


def hold_protected_401(page, target):
    held = []
    page.evaluate("window.protected401Held=false")

    def intercept(route):
        held.append(route)
        route.request.frame.evaluate("window.protected401Held=true")

    page.route(target, intercept, times=1)
    return held

@pytest.mark.parametrize("receiver_host", ["vaultbridge.example.test", "other.example.test"])
def test_native_other_origin_locks_and_journal_do_not_fence_cookie_delivery(named_context, named, receiver_host):
    from starlette.responses import HTMLResponse

    async def receiver(scope, receive, send):
        if scope["type"] == "lifespan":
            while True:
                event = await receive()
                await send({"type": event["type"] + ".complete"})
                if event["type"] == "lifespan.shutdown":
                    return
        else:
            await HTMLResponse("<!doctype html><title>Origin probe</title>", headers={"Connection": "close"})(
                scope, receive, send
            )

    a = open_page(named_context, named)
    login(a)
    clean, cookie = journal(a), session_cookie(named_context)
    with serve(receiver, named["cert"], named["key"]) as port:
        b = named_context.new_page()
        b.goto(f"https://{receiver_host}:{port}/probe")
        assert b.evaluate("keys => keys.map(key=>localStorage.getItem(key))", [INTENT, JOURNAL]) == [None, None]
        delivered_scope = [c for c in named_context.cookies([b.url]) if c["name"] == COOKIE]
        assert bool(delivered_scope) == (receiver_host == "vaultbridge.example.test")
        a.evaluate(
            "lock => {window.originLockHeld=false; navigator.locks.request(lock, async()=>{"
            "originLockHeld=true; await new Promise(resolve=>window.releaseOriginLock=resolve);});}",
            LOCK,
        )
        a.wait_for_function("originLockHeld")
        try:
            assert b.evaluate("lock=>navigator.locks.request(lock,{ifAvailable:true},held=>!!held)", LOCK)
            assert journal(a) == clean and session_cookie(named_context) == cookie
        finally:
            a.evaluate("releaseOriginLock()")
            # Stop native speculative connections before the TLS listener.
            named_context.close()

def test_native_status_readonly_mapping_and_lock_order(settings, backend, monkeypatch):
    settings = policy(settings, "research", indexing="enabled")
    app = create_app(settings=settings)
    runtime = app.state.space_runtime
    with TestClient(app) as client:
        runtime.scheduler.wait(20)
        binding = runtime.registry._binding(SpaceId("research"))
        service = binding._owners.semantic
        assert service.repository.db_path.exists()
        original = SharedSemanticScheduler.observe_status

        def observe(scheduler, selected, cancel=None):
            assert runtime.admission._active_thread is None, "scheduler condition acquired inside admission"
            return original(scheduler, selected, cancel)

        monkeypatch.setattr(SharedSemanticScheduler, "observe_status", observe)

        def forbidden(*args, **kwargs):
            pytest.fail("status invoked rich counts, initialization, model or index work")

        monkeypatch.setattr(SemanticRepository, "read_status", forbidden)
        monkeypatch.setattr(service, "_initialize_state", forbidden)
        monkeypatch.setattr(service, "_resolve_index_signature", forbidden)
        monkeypatch.setattr(service, "_embed", forbidden)
        entry = runtime.scheduler._owners[SpaceId("research")]

        def request():
            before = {
                p: (p.read_bytes(), p.stat().st_mtime_ns) for p in settings.vault_path.parent.rglob("*") if p.is_file()
            }
            response = client.get(
                "/api/v1/knowledge/spaces/status?spaces=research", headers={"Authorization": "<HEADER_VALUE_REDACTED>" + KEY}
            )
            assert response.status_code == 200
            after = {
                p: (p.read_bytes(), p.stat().st_mtime_ns) for p in settings.vault_path.parent.rglob("*") if p.is_file()
            }
            assert after == before
            return response.json()["spaces"][0]

        assert request() == dict(
            space_id="research", index_state="ready", search_available=True, refresh_required=False
        )
        with runtime.scheduler._condition:
            entry.pending.add("not-inspected.md")
        assert request() == dict(
            space_id="research", index_state="indexing", search_available=True, refresh_required=True
        )
        with runtime.scheduler._condition:
            entry.paused = True
            entry.error_type = "SyntheticFailure"
        assert request() == dict(
            space_id="research", index_state="unavailable", search_available=True, refresh_required=True
        )
        with runtime.scheduler._condition:
            entry.pending.clear()
            entry.paused = False
            entry.error_type = None
            entry.coverage_revision += 1
        assert request() == dict(
            space_id="research", index_state="unavailable", search_available=True, refresh_required=True
        )
        # A compatible previous index remains searchable, without claiming ready.
        service.repository.set_metadata("index_state", "error")
        assert request() == dict(
            space_id="research", index_state="unavailable", search_available=True, refresh_required=True
        )
        # Binding failure is safe unavailable, never an alternate-space probe.
        service.repository.db_path.unlink()
        assert request() == dict(
            space_id="research", index_state="unavailable", search_available=False, refresh_required=True
        )
        assert runtime.admission._active_thread != threading.get_ident()

def test_scheduler_observation_wait_is_cancellable_before_admission(settings, backend, monkeypatch):

        monkeypatch.setattr(SharedSemanticScheduler, "observe_status", observe)

                )

                assert runtime.admission._active_thread is None
